<script>
(()=>{
  const buttons=[...document.querySelectorAll('.filter')];
  const cards=[...document.querySelectorAll('.card')];
  const count=document.getElementById('projectCount');
  const modal=document.getElementById('projectModal');
  const modalImg=document.getElementById('modalImg');
  const modalTitle=document.getElementById('modalTitle');
  const modalDescription=document.getElementById('modalDescription');
  const modalCategory=document.getElementById('modalCategory');
  const modalService=document.getElementById('modalService');
  const close=document.getElementById('modalClose');

  const updateCount=()=>{
    const n=cards.filter(card=>!card.hidden).length;
    count.textContent=`${n} ${n===1?'proyecto':'proyectos'} seleccionados`;
  };

  buttons.forEach(btn=>btn.addEventListener('click',()=>{
    const filter=btn.dataset.filter;
    buttons.forEach(x=>x.classList.toggle('active',x===btn));
    cards.forEach(card=>card.hidden=filter!=='all'&&card.dataset.category!==filter);
    updateCount();
  }));

  const openModal=card=>{
    modalImg.src=card.dataset.image||'';
    modalImg.alt=card.dataset.title||'Proyecto';
    modalTitle.textContent=card.dataset.title||'Proyecto';
    modalDescription.textContent=card.dataset.description||'';
    const labels={arquitectura:'Arquitectura',grafico:'Diseño Gráfico',web:'Diseño Web'};
    modalCategory.textContent=labels[card.dataset.category]||'Proyecto';
    modalService.href=card.dataset.service||'/';
    modal.classList.add('show');
    modal.setAttribute('aria-hidden','false');
    document.documentElement.style.overflow='hidden';
  };
  const closeModal=()=>{
    modal.classList.remove('show');
    modal.setAttribute('aria-hidden','true');
    document.documentElement.style.overflow='';
  };
  cards.forEach(card=>{
    card.tabIndex=0;
    card.setAttribute('role','button');
    card.addEventListener('click',()=>openModal(card));
    card.addEventListener('keydown',e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();openModal(card)}});
  });
  close.addEventListener('click',closeModal);
  modal.addEventListener('click',e=>{if(e.target===modal)closeModal()});
  document.addEventListener('keydown',e=>{if(e.key==='Escape')closeModal()});
})();

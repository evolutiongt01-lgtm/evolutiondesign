(function(){
  const c=document.getElementById('heroCanvas');
  if(!c||matchMedia('(prefers-reduced-motion:reduce)').matches)return;

  const ctx=c.getContext('2d',{alpha:true});
  const glyphs=[
    '0','1','<','>','/','{','}','[',']','(',')',';','=','+','*','#','&',
    'const','let','=>','API','UI','UX','JS','CSS','div','async','await','fetch'
  ];

  let w=0,h=0,dpr=1,columns=[],last=0;

  function makeColumn(x,gap){
    const mobile=innerWidth<700;
    return {
      x,
      y:-(Math.random()*h),
      speed:(mobile?0.75:0.9)+Math.random()*(mobile?1.15:1.8),
      size:(mobile?10:10.5)+Math.random()*(mobile?2:3.5),
      alpha:.55+Math.random()*.20,
      gap,
      token:glyphs[(Math.random()*glyphs.length)|0],
      swap:0
    };
  }

  function resize(){
    w=c.clientWidth;
    h=c.clientHeight;
    dpr=Math.min(window.devicePixelRatio||1,innerWidth<700?1.15:1.65);
    c.width=Math.max(1,Math.floor(w*dpr));
    c.height=Math.max(1,Math.floor(h*dpr));
    ctx.setTransform(dpr,0,0,dpr,0,0);

    const gap=innerWidth<700?38:42;
    const count=Math.ceil(w/gap)+1;
    columns=Array.from({length:count},(_,i)=>makeColumn(i*gap+(Math.random()*8-4),gap));
    ctx.clearRect(0,0,w,h);
  }

  function drawToken(col,y,alpha,head=false){
    const leftQuiet=col.x<w*.52 ? .55 : 1;
    ctx.font=`${head?600:400} ${col.size}px ui-monospace,SFMono-Regular,Menlo,Monaco,Consolas,monospace`;
    ctx.fillStyle=`rgba(255,255,255,${Math.max(0,alpha*leftQuiet)})`;
    ctx.shadowColor='rgba(255,255,255,.18)';
    ctx.shadowBlur=(innerWidth<700?0:(head?4:1));
    ctx.fillText(col.token,col.x,y);
  }

  function frame(t){
    requestAnimationFrame(frame);
    const frameGap=innerWidth<700?48:34;
    if(t-last<frameGap)return;
    last=t;

    /* Desvanece la estela sin pintar un rectángulo negro sobre el hero. */
    ctx.save();
    ctx.globalCompositeOperation='destination-out';
    ctx.fillStyle='rgba(0,0,0,.115)';
    ctx.fillRect(0,0,w,h);
    ctx.restore();

    for(const col of columns){
      col.y+=col.speed*3.2;
      col.swap++;

      if(col.swap>(innerWidth<700?5:4)){
        col.swap=0;
        if(Math.random()<.58) col.token=glyphs[(Math.random()*glyphs.length)|0];
      }

      const line=col.size*1.55;
      drawToken(col,col.y,col.alpha,true);

      for(let trail=1;trail<=5;trail++){
        const ty=col.y-(trail*line);
        if(ty>-20){
          drawToken(col,ty,col.alpha*(.54-trail*.075),false);
        }
      }

      if(col.y>h+80){
        col.y=-(30+Math.random()*h*.42);
        col.speed=(innerWidth<700?.75:.9)+Math.random()*(innerWidth<700?1.15:1.8);
        col.alpha=.55+Math.random()*.20;
        col.token=glyphs[(Math.random()*glyphs.length)|0];
      }
    }
  }

  resize();
  addEventListener('resize',resize,{passive:true});
  requestAnimationFrame(frame);
})();

function loadEvolutionRecaptchaEnterprise(siteKey) {
  if (window.grecaptcha?.enterprise) return Promise.resolve();
  if (evolutionRecaptchaScriptPromise) return evolutionRecaptchaScriptPromise;

  evolutionRecaptchaScriptPromise = new Promise((resolve, reject) => {
    const existing = document.querySelector('script[data-evolution-recaptcha-enterprise]');
    if (existing) {
      if (window.grecaptcha?.enterprise) { resolve(); return; }
      existing.addEventListener('load', () => resolve(), { once: true });
      existing.addEventListener('error', () => reject(new Error('No se pudo cargar reCAPTCHA Enterprise.')), { once: true });
      return;
    }

    const script = document.createElement('script');
    script.src = `https://www.google.com/recaptcha/enterprise.js?render=${encodeURIComponent(siteKey)}`;
    script.async = true;
    script.defer = true;
    script.dataset.evolutionRecaptchaEnterprise = 'true';
    script.onload = () => resolve();
    script.onerror = () => reject(new Error('No se pudo cargar reCAPTCHA Enterprise.'));
    document.head.appendChild(script);
  }).catch(error => {
    evolutionRecaptchaScriptPromise = null;
    throw error;
  });

  return evolutionRecaptchaScriptPromise;
}

  return new Promise((resolve, reject) => {
    window.grecaptcha.enterprise.ready(async () => {
      try {

        if (!token) throw new Error('reCAPTCHA no devolvió un token válido.');
        resolve(token);

        reject(error);
      }
    });
  });
}

async function compressImageToDataUrl(file,maxSide=1100,quality=.78,maxChars=560000){
  const img=await imageFromFile(file);let scale=Math.min(1,maxSide/Math.max(img.naturalWidth||1,img.naturalHeight||1));
  for(let tries=0;tries<5;tries++){
    const c=document.createElement('canvas');c.width=Math.max(1,Math.round(img.naturalWidth*scale));c.height=Math.max(1,Math.round(img.naturalHeight*scale));
    const x=c.getContext('2d',{alpha:true});x.drawImage(img,0,0,c.width,c.height);
    const data=c.toDataURL('image/webp',Math.max(.48,quality-tries*.08));
    if(data.length<=maxChars)return data;scale*=.78;
  }
  throw new Error('La imagen sigue siendo demasiado pesada. Usa una captura o imagen más ligera.');
}
